```rust
use std::path::Path;
use chrono::NaiveDate;
use crate::note::{Note, NoteMetadata};

pub fn parse_note(path: &Path, content: &str) -> Note {
    let title = extract_title(path, content);
    let links = extract_links(content);
    let tags = extract_tags(content);
    let dates = extract_dates(content);

    Note {
        path: path.to_path_buf(),
        metadata: NoteMetadata { title, links, tags, dates },
    }
}

fn extract_title(path: &Path, content: &str) -> String {
    content.lines()
        .find_map(|l| l.strip_prefix("# "))
        .map(|t| t.trim().to_string())
        .unwrap_or_else(|| {
            path.file_stem()
                .and_then(|s| s.to_str())
                .unwrap_or("Untitled")
                .to_string()
        })
}

fn extract_links(content: &str) -> Vec<String> {
    let re = regex::Regex::new(r"\[\[(.+?)(?:\|.+?)?\]\]").unwrap();
    re.captures_iter(content)
        .map(|c| c[1].to_string())
        .collect()
}

fn extract_tags(content: &str) -> Vec<String> {
    let re = regex::Regex::new(r"(?m:^|\s)#([a-zA-Zа-яА-Я0-9_\-]+)").unwrap();
    let mut tags: Vec<String> = re.captures_iter(content)
        .map(|c| c[1].to_string())
        .collect();
    tags.sort();
    tags.dedup();
    tags
}

fn extract_dates(content: &str) -> Vec<NaiveDate> {
    let re = regex::Regex::new(r"\b(\d{2})\.(\d{2})\.(\d{4})\b").unwrap();
    re.captures_iter(content)
        .filter_map(|c| {
            let day = c[1].parse::<u32>().ok()?;
            let month = c[2].parse::<u32>().ok()?;
            let year = c[3].parse::<i32>().ok()?;
            NaiveDate::from_ymd_opt(year, month, day)
        })
        .collect()
}
```

```rust
use std::path::PathBuf;
use chrono::NaiveDate;
use crate::index::ConcurrentIndex;

#[derive(Debug, Clone)]
pub enum Query {
    TagsContain(String),
    DateBefore(NaiveDate),
    DateAfter(NaiveDate),
    Text(String),
    And(Box<Query>, Box<Query>),
    Or(Box<Query>, Box<Query>),
}

pub fn parse_query(input: &str) -> Result<Query, String> {
    let input = input.trim();

    // Split on " and " / " or "  (simple top-level split)
    if let Some(pos) = input.find(" and ") {
        let before = &input[..pos];
        let after = &input[pos + 5..];
        return Ok(Query::And(
            Box::new(parse_query(before)?),
            Box::new(parse_query(after)?),
        ));
    }

    if let Some(pos) = input.find(" or ") {
        let before = &input[..pos];
        let after = &input[pos + 4..];
        return Ok(Query::Or(
            Box::new(parse_query(before)?),
            Box::new(parse_query(after)?),
        ));
    }

    if input.starts_with("tags contain ") {
        let tag = input.trim_start_matches("tags contain ");
        let tag = tag.trim_matches('"').to_string();
        return Ok(Query::TagsContain(tag));
    }

    if input.starts_with("date before ") {
        let date_str = input.trim_start_matches("date before ").trim();
        let date = NaiveDate::parse_from_str(date_str, "%d.%m.%Y")
            .map_err(|e| format!("Invalid date format: {}", e))?;
        return Ok(Query::DateBefore(date));
    }

    if input.starts_with("date after ") {
        let date_str = input.trim_start_matches("date after ").trim();
        let date = NaiveDate::parse_from_str(date_str, "%d.%m.%Y")
            .map_err(|e| format!("Invalid date format: {}", e))?;
        return Ok(Query::DateAfter(date));
    }

    // Plain text search
    Ok(Query::Text(input.to_string()))
}

#[derive(Debug)]
pub struct SearchResult {
    pub path: PathBuf,
    pub title: String,
}

pub fn execute_query(index: &ConcurrentIndex, query: &Query) -> Vec<PathBuf> {
    match query {
        Query::TagsContain(tag) => index.tags.get(tag),
        Query::DateBefore(date) => {
            index.dates.all_dates().into_iter()
                .filter(|(d, _)| d <= date)
                .flat_map(|(_, paths)| paths)
                .collect()
        }
        Query::DateAfter(date) => {
            index.dates.all_dates().into_iter()
                .filter(|(d, _)| d >= date)
                .flat_map(|(_, paths)| paths)
                .collect()
        }
        Query::Text(term) => index.fulltext.search(term),
        Query::And(left, right) => {
            let left_results: std::collections::HashSet<_> =
                execute_query(index, left).into_iter().collect();
            let right_results: std::collections::HashSet<_> =
                execute_query(index, right).into_iter().collect();
            left_results.intersection(&right_results).cloned().collect()
        }
        Query::Or(left, right) => {
            let mut results = execute_query(index, left);
            results.extend(execute_query(index, right));
            results.sort();
            results.dedup();
            results
        }
    }
}
```

```rust
use std::path::Path;
use std::sync::mpsc;
use std::thread;
use notify::{Config, Event, EventKind, RecommendedWatcher, RecursiveMode, Watcher as NotifyWatcher};

#[derive(Debug, Clone)]
pub enum FileEvent {
    Created(String),
    Modified(String),
    Deleted(String),
}

pub struct FileWatcher {
    pub receiver: mpsc::Receiver<FileEvent>,
    _handle: thread::JoinHandle<()>,
}

impl FileWatcher {
    pub fn start(path: &Path) -> Result<Self, String> {
        let (tx, rx) = mpsc::channel();
        let watch_path = path.to_path_buf();

        let handle = thread::spawn(move || {
            let (notify_tx, notify_rx) = mpsc::channel();

            let mut watcher = RecommendedWatcher::new(
                move |res: Result<Event, notify::Error>| {
                    if let Ok(event) = res {
                        notify_tx.send(event).ok();
                    }
                },
                Config::default(),
            ).expect("Failed to create watcher");

            watcher.watch(&watch_path, RecursiveMode::Recursive)
                .expect("Failed to watch path");

            for event in notify_rx {
                for path in event.paths {
                    if path.starts_with(watch_path.join(".git"))
                        || path.starts_with(watch_path.join(".index"))
                    {
                        continue;
                    }

                    let event = match event.kind {
                        EventKind::Create(_) => FileEvent::Created(path.to_string_lossy().to_string()),
                        EventKind::Modify(_) => FileEvent::Modified(path.to_string_lossy().to_string()),
                        EventKind::Remove(_) => FileEvent::Deleted(path.to_string_lossy().to_string()),
                        _ => continue,
                    };

                    if tx.send(event).is_err() {
                        break;
                    }
                }
            }
        });

        Ok(Self {
            receiver: rx,
            _handle: handle,
        })
    }
}
```

```rust
use std::path::{Path, PathBuf};
use std::sync::Arc;
use std::sync::mpsc;
use std::thread;
use dashmap::DashMap;
use crate::index::ConcurrentIndex;
use crate::note::Note;
use crate::parser;
use crate::search::{self, Query, SearchResult};
use crate::watcher::{FileEvent, FileWatcher};

pub struct Vault {
    pub path: PathBuf,
    pub index: Arc<ConcurrentIndex>,
    debounce_rx: mpsc::Receiver<()>,
}

#[derive(Debug)]
pub struct IndexReport {
    pub total_notes: usize,
    pub total_tags: usize,
    pub total_dates: usize,
}

impl Vault {
    pub fn open(path: &Path) -> Result<Self, String> {
        let path = path.to_path_buf();

        if !path.exists() {
            return Err(format!("Path does not exist: {}", path.display()));
        }

        let index = Arc::new(ConcurrentIndex::new());

        let vault = Self {
            path,
            index,
            debounce_rx: mpsc::channel().1,
        };

        vault.rebuild_index();

        Ok(vault)
    }

    pub fn rebuild_index(&self) {
        let index = self.index.clone();
        let vault_path = self.path.clone();

        thread::spawn(move || {
            index.clear();

            let md_files: Vec<_> = walkdir(&vault_path)
                .filter(|p| p.extension().map(|e| e == "md").unwrap_or(false))
                .collect();

            let notes: Vec<Note> = md_files.par_iter()
                .filter_map(|path| {
                    std::fs::read_to_string(path).ok().map(|content| {
                        parser::parse_note(path, &content)
                    })
                })
                .collect();

            for note in &notes {
                index.index_note(note);
            }
        });
    }

    pub fn search(&self, query_str: &str) -> Result<Vec<SearchResult>, String> {
        let query = search::parse_query(query_str)?;
        let paths = search::execute_query(&self.index, &query);

        let results = paths.into_iter()
            .map(|path| {
                let title = path.file_stem()
                    .and_then(|s| s.to_str())
                    .unwrap_or("Untitled")
                    .to_string();
                SearchResult { path, title }
            })
            .collect();

        Ok(results)
    }

    pub fn get_note(&self, path: &Path) -> Result<String, String> {
        let full_path = self.path.join(path);
        std::fs::read_to_string(&full_path)
            .map_err(|e| format!("Failed to read {}: {}", full_path.display(), e))
    }

    pub fn write_note(&self, path: &Path, content: &str) -> Result<(), String> {
        let full_path = self.path.join(path);

        // Create parent directories if needed
        if let Some(parent) = full_path.parent() {
            std::fs::create_dir_all(parent)
                .map_err(|e| format!("Failed to create directories: {}", e))?;
        }

        std::fs::write(&full_path, content)
            .map_err(|e| format!("Failed to write {}: {}", full_path.display(), e))?;

        Ok(())
    }

    pub fn get_all_tags(&self) -> Vec<String> {
        self.index.tags.all_tags()
    }

    pub fn get_all_dates(&self) -> Vec<(chrono::NaiveDate, Vec<PathBuf>)> {
        self.index.dates.all_dates()
    }

    pub fn validate_indexes(&self) -> IndexReport {
        IndexReport {
            total_notes: self.index.file_states.len(),
            total_tags: self.index.tags.all_tags().len(),
            total_dates: self.index.dates.all_dates().len(),
        }
    }
}

fn walkdir(path: &Path) -> Vec<PathBuf> {
    let mut files = Vec::new();
    if let Ok(entries) = std::fs::read_dir(path) {
        for entry in entries.flatten() {
            let entry_path = entry.path();
            if entry_path.is_dir() {
                // Skip .git and .index
                if entry_path.file_name()
                    .and_then(|n| n.to_str())
                    .map(|n| n == ".git" || n == ".index")
                    .unwrap_or(false)
                {
                    continue;
                }
                files.extend(walkdir(&entry_path));
            } else {
                files.push(entry_path);
            }
        }
    }
    files
}
```

```rust
use std::path::Path;

pub struct GitBackend {
    repo_path: std::path::PathBuf,
}

impl GitBackend {
    pub fn open(path: &Path) -> Result<Self, String> {
        if !path.join(".git").exists() {
            return Err("Not a git repository".to_string());
        }
        Ok(Self { repo_path: path.to_path_buf() })
    }

    pub fn auto_commit(&self, message: &str) -> Result<(), String> {
        let repo = git2::Repository::open(&self.repo_path)
            .map_err(|e| format!("Failed to open repo: {}", e))?;

        let mut index = repo.index()
            .map_err(|e| format!("Failed to open index: {}", e))?;

        index.add_all(["*"].iter(), git2::IndexAddOption::DEFAULT, None)
            .map_err(|e| format!("Failed to add files: {}", e))?;

        index.write()
            .map_err(|e| format!("Failed to write index: {}", e))?;

        let oid = index.write_tree()
            .map_err(|e| format!("Failed to write tree: {}", e))?;

        let tree = repo.find_tree(oid)
            .map_err(|e| format!("Failed to find tree: {}", e))?;

        let author = git2::Signature::now("Simpler Notes", "notes@simpler.app")
            .map_err(|e| format!("Failed to create signature: {}", e))?;

        let parent = repo.head()
            .ok()
            .and_then(|head| head.target())
            .and_then(|oid| repo.find_commit(oid).ok());

        let _commit = match parent {
            Some(parent) => repo.commit(
                Some("HEAD"),
                &author,
                &author,
                message,
                &tree,
                &[&parent],
            ),
            None => repo.commit(
                Some("HEAD"),
                &author,
                &author,
                message,
                &tree,
                &[] as &[&git2::Commit],
            ),
        }.map_err(|e| format!("Failed to commit: {}", e))?;

        Ok(())
    }

    pub fn push(&self) -> Result<(), String> {
        let repo = git2::Repository::open(&self.repo_path)
            .map_err(|e| format!("Failed to open repo: {}", e))?;

        let mut remote = repo.find_remote("origin")
            .map_err(|_| "No remote 'origin' configured".to_string())?;

        remote.push(&["refs/heads/main"], None)
            .map_err(|e| format!("Failed to push: {}", e))?;

        Ok(())
    }

    pub fn pull(&self) -> Result<(), String> {
        let repo = git2::Repository::open(&self.repo_path)
            .map_err(|e| format!("Failed to open repo: {}", e))?;

        let mut remote = repo.find_remote("origin")
            .map_err(|_| "No remote 'origin' configured".to_string())?;

        remote.fetch(&["main"], None, None)
            .map_err(|e| format!("Failed to fetch: {}", e))?;

        let fetch_head = repo.find_reference("FETCH_HEAD")
            .map_err(|e| format!("Failed to find FETCH_HEAD: {}", e))?;

        let fetch_commit = repo.reference_to_annotated_commit(&fetch_head)
            .map_err(|e| format!("Failed to resolve FETCH_HEAD: {}", e))?;

        let analysis = repo.merge_analysis(&[&fetch_commit])
            .map_err(|e| format!("Failed to merge analysis: {}", e))?;

        if analysis.0.is_up_to_date() {
            return Ok(());
        }

        let refname = "refs/heads/main";
        if let Ok(branch_ref) = repo.find_reference(refname) {
            let branch_oid = branch_ref.target().unwrap();
            let branch_commit = repo.find_commit(branch_oid)
                .map_err(|e| format!("Failed to find branch commit: {}", e))?;

            repo.merge(&[&fetch_commit], None, None)
                .map_err(|e| format!("Failed to merge: {}", e))?;

            // Check for conflicts
            if repo.index().map(|i| i.has_conflicts()).unwrap_or(false) {
                return Err("Merge conflicts detected. Resolve manually.".to_string());
            }

            let tree_oid = repo.index()
                .and_then(|mut i| i.write_tree())
                .map_err(|e| format!("Failed to write merge tree: {}", e))?;

            let tree = repo.find_tree(tree_oid)
                .map_err(|e| format!("Failed to find merge tree: {}", e))?;

            let author = git2::Signature::now("Simpler Notes", "notes@simpler.app")
                .map_err(|e| format!("Failed to create signature: {}", e))?;

            let _ = repo.commit(
                Some(refname),
                &author,
                &author,
                "Merge remote-tracking branch 'origin/main'",
                &tree,
                &[&branch_commit, &fetch_commit],
            );
        }

        Ok(())
    }

    pub fn status(&self) -> Result<String, String> {
        let repo = git2::Repository::open(&self.repo_path)
            .map_err(|e| format!("Failed to open repo: {}", e))?;

        let mut statuses = repo.statuses(None)
            .map_err(|e| format!("Failed to get statuses: {}", e))?;

        let clean = statuses.iter().all(|s| s.status() == git2::Status::CURRENT);

        Ok(if clean {
            "clean".to_string()
        } else {
            format!("{} dirty file(s)", statuses.len())
        })
    }
}
```

```rust
impl ConcurrentIndex {
    pub fn save(&self, path: &Path) -> Result<(), String> {
        fs::create_dir_all(path)
            .map_err(|e| format!("Failed to create index dir: {}", e))?;

        // Serialize tag_index
        let tags_path = path.join("tags.json");
        let tag_data: Vec<(String, Vec<PathBuf>)> = self.tags.all_tags().into_iter()
            .map(|tag| {
                let paths = self.tags.get(&tag);
                (tag, paths)
            })
            .collect();
        fs::write(&tags_path, serde_json::to_string_pretty(&tag_data)
            .map_err(|e| format!("Failed to serialize tags: {}", e))?)
            .map_err(|e| format!("Failed to write tags: {}", e))?;

        // Serialize date_index
        let dates_path = path.join("dates.json");
        fs::write(&dates_path, serde_json::to_string_pretty(
            &self.dates.all_dates()
        ).map_err(|e| format!("Failed to serialize dates: {}", e))?)
            .map_err(|e| format!("Failed to write dates: {}", e))?;

        Ok(())
    }

    pub fn load(path: &Path) -> Result<Self, String> {
        let index = Self::new();

        // Load tags
        let tags_path = path.join("tags.json");
        if tags_path.exists() {
            let data = fs::read_to_string(&tags_path)
                .map_err(|e| format!("Failed to read tags: {}", e))?;
            let tag_data: Vec<(String, Vec<PathBuf>)> = serde_json::from_str(&data)
                .map_err(|e| format!("Failed to parse tags: {}", e))?;
            for (tag, paths) in tag_data {
                for path in paths {
                    index.tags.add(&tag, path);
                }
            }
        }

        // Load dates
        let dates_path = path.join("dates.json");
        if dates_path.exists() {
            let data = fs::read_to_string(&dates_path)
                .map_err(|e| format!("Failed to read dates: {}", e))?;
            let date_data: Vec<(chrono::NaiveDate, Vec<PathBuf>)> = serde_json::from_str(&data)
                .map_err(|e| format!("Failed to parse dates: {}", e))?;
            for (date, paths) in date_data {
                for path in paths {
                    index.dates.add(date, path);
                }
            }
        }

        Ok(index)
    }
```

```rust
use crate::index::ConcurrentIndex;  // already present

impl Vault {
    pub fn open(path: &Path) -> Result<Self, String> {
        let path = path.to_path_buf();
        if !path.exists() {
            return Err(format!("Path does not exist: {}", path.display()));
        }

        let index_dir = path.join(".index");
        let index = if index_dir.exists() {
            Arc::new(ConcurrentIndex::load(&index_dir).unwrap_or_default())
        } else {
            Arc::new(ConcurrentIndex::new())
        };

        let vault = Self { path, index, debounce_rx: mpsc::channel().1 };
        vault.rebuild_index();
        Ok(vault)
    }

    pub fn rebuild_index(&self) {
        let index = self.index.clone();
        let vault_path = self.path.clone();

        thread::spawn(move || {
            index.clear();

            let md_files: Vec<_> = walkdir(&vault_path)
                .filter(|p| p.extension().map(|e| e == "md").unwrap_or(false))
                .collect();

            let notes: Vec<Note> = md_files.par_iter()
                .filter_map(|path| {
                    std::fs::read_to_string(path).ok().map(|content| {
                        parser::parse_note(path, &content)
                    })
                })
                .collect();

            for note in &notes {
                index.index_note(note);
            }

            // Persist after rebuild
            let _ = index.save(&vault_path.join(".index"));
        });
    }
}
```
